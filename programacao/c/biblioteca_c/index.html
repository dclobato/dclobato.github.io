/**
 * @brief Lê um número inteiro da entrada padrão.
 *
 * Esta função solicita ao usuário a entrada de um valor inteiro através
 * do prompt fornecido. A função continuará solicitando a entrada até que
 * um valor inteiro válido seja fornecido, dentro dos limites do tipo int.
 *
 * @param prompt Mensagem exibida ao usuário solicitando a entrada (pode
 *        ser NULL)
 * @return O valor inteiro lido da entrada padrão
 */
int lerInt(const char *prompt)
{
  char *linha = NULL;
  char *p     = NULL;
  long  valor = 0L;

  while (1)
  {
    if (prompt != NULL)
    {
      printf("%s", prompt);
    }
    linha = lerLinha(NULL);
    valor = strtol(linha, &p, 10);
    if ((p == linha) || (errno == ERANGE) || (*p && *p != '\n'))
    {
      free(linha);
      continue;
    }
    if (valor < INT_MIN || valor > INT_MAX)
    {
      free(linha);
      continue;
    }
    break;
  }
  free(linha);
  return (int) valor;
}

/**
 * @brief Lê uma linha de texto da entrada padrão.
 *
 * Esta função lê uma linha completa de texto da entrada padrão
 * (stdin), até encontrar um caractere de nova linha ou EOF. A memória
 * para armazenar a linha é alocada dinamicamente.
 *
 * @param prompt Mensagem exibida ao usuário solicitando a entrada (pode
 *        ser NULL)
 * @return Ponteiro para a string lida (deve ser liberada pelo chamador
 *         usando free())
 */
char *lerLinha(const char *prompt)
{
  char *line = NULL;
  if (prompt != NULL)
  {
    printf("%s", prompt);
  }
  line = lerLinhaArquivo(stdin);
  return line;
}

/**
 * @brief Lê uma linha de texto de um arquivo especificado.
 *
 * Esta função lê uma linha completa de texto do arquivo especificado,
 * até encontrar um caractere de nova linha ou EOF. A memória para
 * armazenar a linha é alocada dinamicamente. A função gerencia
 * automaticamente o redimensionamento do buffer se necessário.
 *
 * @param arquivo Ponteiro para o arquivo de onde a linha será lida
 * @return Ponteiro para a string lida (deve ser liberada pelo chamador
 *         usando free()), ou NULL se EOF for encontrado sem ler nenhum
 *         caractere
 */
char *lerLinhaArquivo(FILE *arquivo)
{
  char *       line  = NULL;
  char *       nline = NULL;
  unsigned int n     = 0;
  unsigned int size  = 4096;
  int          ch    = 0;

  line = (char *) malloc(size + 1);
  if (line == NULL)
  {
    printf("Sem memoria\n");
    exit(1);
  }
  while ((ch = getc(arquivo)) != '\n' && ch != EOF)
  {
    if (n == size)
    {
      size += size;
      nline = (char *) malloc(size + 1);
      if (nline == NULL)
      {
        free(line);
        printf("Sem memoria\n");
        exit(1);
      }
      strncpy(nline, line, n);
      free(line);
      line = nline;
    }
    line[n++] = (char) ch;
  }
  if (n == 0 && ch == EOF)
  {
    free(line);
    return (NULL);
  }
  line[n] = '\0';
  nline   = (char *) malloc(n + 1);
  if (nline == NULL)
  {
    free(line);
    printf("Sem memoria");
    exit(1);
  }
  strcpy(nline, line);
  free(line);
  return nline;
}

/**
 * @brief Remove espaços em branco do início e do fim de uma string.
 * 
 * Esta função remove todos os caracteres de espaço em branco (conforme
 * definido por isspace()) do início e do fim da string fornecida. A 
 * string é modificada in-place, ou seja, a string original é alterada
 * diretamente.
 *
 * @param[in,out] str Ponteiro para a string a ser processada. A string
 *                    deve ser terminada com o caractere nulo '\0'. Após
 *                    a execução, a string conterá o resultado sem
 *                    espaços em branco nas extremidades.
 * @note A função utiliza memmove() para garantir uma manipulação segura
 *       da memória quando há sobreposição entre as regiões de origem e
 *       destino.
 * @warning A string de entrada deve ser um buffer válido e modificável.
 *          Não deve ser uma string literal ou alocada em região de
 *          memória somente leitura.
 * @see isspace()
 * @see memmove()
 * @see strlen()
 */
void trimString(char *str) {
    int start = 0, end = strlen(str) - 1;

    // Remove espaços no início
    while (isspace((unsigned char)str[start]))
        start++;

    // Remove espaços no final
    while ((end >= start) && isspace((unsigned char)str[end]))
        end--;

    // Move os caracteres para o início
    memmove(str, str + start, end - start + 1);
    str[end - start + 1] = '\0';
}

/**
 * @brief Inicializa o gerador de números aleatórios.
 *
 * Esta função inicializa o gerador de números pseudoaleatórios
 * utilizando o tempo atual como semente. Deve ser chamada uma única
 * vez no início do programa antes de qualquer chamada às funções
 * RandomInteger ou RandomReal.
 */
void Randomize(void)
{
  srand((int) time(NULL));
}

/**
 * @brief Gera um número inteiro aleatório dentro de um intervalo especificado.
 *
 * Esta função retorna um número inteiro pseudoaleatório uniformemente
 * distribuído no intervalo [low, high], inclusive. Requer que a função
 * Randomize() tenha sido chamada previamente para inicializar o gerador
 * de números aleatórios.
 *
 * @param low Limite inferior do intervalo
 * @param high Limite superior do intervalo
 * @return Um número inteiro aleatório entre low e high, inclusive
 */
int RandomInteger(long int low, long int high)
{
  long int range;
  long int value;

  if (low < high)
  {
    t = low;
    low = high;
    high = t;
  }

  range = high - low + 1;
  value = rand() % range;  // Pode gerar um viés de módulo, mas para a
                           // maioria dos nossos casos, está Ok  
  return low + value;
}

/**
 * @brief Gera um número real aleatório dentro de um intervalo especificado.
 *
 * Esta função retorna um número de ponto flutuante pseudoaleatório
 * uniformemente distribuído no intervalo [low, high). Note que o limite
 * superior (high) não está incluído no intervalo de valores possíveis.
 * Requer que a função Randomize() tenha sido chamada previamente para
 * inicializar o gerador de números aleatórios.
 *
 * @param low Limite inferior do intervalo (incluído)
 * @param high Limite superior do intervalo (não incluído)
 * @return Um número real aleatório entre low (inclusive) e high (exclusive)
 */
double RandomReal(double low, double high)
{
  double d;

  d = (double) rand() / ((double) RAND_MAX + 1);
  return (low + d * (high - low));
}

    Funções trigonométricas¶

    • double sin(double x);: Retorna o seno de x (em radianos);
    • double cos(double x);: Retorna o cosseno de x (em radianos);
    • double tan(double x);: Retorna a tangente de x (em radianos);
    • double asin(double x);: Retorna o arco-seno de x;
    • double acos(double x);: Retorna o arco-cosseno de x;
    • double atan(double x);: Retorna o arco-tangente de x;
    • double atan2(double y, double x);: Retorna o arco-tangente de y/x, considerando o quadrante correto.

    Funções exponenciais e logarítmicas¶

    • double exp(double x);: Retorna \(e^x\);
    • double log(double x);: Retorna o logaritmo base \(e\) de x;
    • double log10(double x);: Retorna o logaritmo base 10 de x;
    • double log2(double x);: Retorna o logaritmo base 2 de x.

    Funções de arredondamento e manipulação de ponto flutuante¶

    • double ceil(double x);: Retorna o menor inteiro maior ou igual a x;
    • double floor(double x);: Retorna o maior inteiro menor ou igual a x;
    • double round(double x);: Retorna x arredondado para o inteiro mais próximo;
    • double trunc(double x);: Remove a parte decimal de x.

    string.h¶

    A biblioteca string.h fornece funções para manipulação de cadeia de caracteres.

    Funções de cópia e concatenação¶

    • char *strcpy(char *dest, const char *src);: Copia src para dest;
    • char *strncpy(char *dest, const char *src, size_t n);: Copia até n caracteres de src para dest;
    • char *strcat(char *dest, const char *src);: Concatena src ao final de dest;
    • char *strncat(char *dest, const char *src, size_t n);: Concatena até n caracteres de src a dest.

    Funções de comparação¶

    • int strcmp(const char *s1, const char *s2);: Compara s1 e s2, retornando <0, 0 ou >0;
    • int strncmp(const char *s1, const char *s2, size_t n);: Compara até n caracteres de s1 e s2.

    Funções de tamanho e busca¶

    • size_t strlen(const char *s);: Retorna o comprimento de s (sem contar \0);
    • char *strchr(const char *s, int c);: Retorna um ponteiro para a primeira ocorrência de c em s;
    • char *strrchr(const char *s, int c);: Retorna um ponteiro para a última ocorrência de c em s;
    • char *strstr(const char *palheiro, const char *agulha);: Retorna um ponteiro para a primeira ocorrência de agulha dentro de palheiro;
    • char *strtok(char *str, const char *delim);: Divide str em tokens usando delim como delimitador.

    Funções de manipulação de memória¶

    • void *memcpy(void *dest, const void *src, size_t n);: Copia n bytes de src para dest;
    • void *memmove(void *dest, const void *src, size_t n);: Copia n bytes, tratando sobreposição de memória;
    • void *memset(void *s, int c, size_t n);: Preenche os n primeiros bytes de s com o valor c;
    • int memcmp(const void *s1, const void *s2, size_t n);: Compara n bytes de s1 e s2.

    ctype.h¶

    A biblioteca <ctype.h> fornece funções para classificação e manipulação de caracteres em C. Essas funções operam sobre char e int, permitindo identificar e converter caracteres de maneira eficiente, o que é essencial para validar e manipular caracteres em entrada de usuários, formatação de textos e análise de dados.

    Funções de classificação de caracteres¶

    • int isalpha(int c): Retorna verdadeiro (!= 0) se c for uma letra (a-z ou A-Z);
    • int isdigit(int c): Retorna verdadeiro se c for um dígito (0-9);
    • int isalnum(int c): Retorna verdadeiro se c for uma letra (a-z, A-Z) ou um número (0-9);
    • int isspace(int c): Retorna verdadeiro se c for um espaço em branco (' ', '\t', '\n', '\r', '\v', '\f');
    • int ispunct(int c): Retorna verdadeiro se c for um caractere de pontuação (ex: !@#$%^&*);
    • int isprint(int c): Retorna verdadeiro se c for um caractere imprimível (excluindo '\0' e caracteres de controle);
    • int isgraph(int c): Igual a isprint(), mas exclui espaços em branco;
    • int iscntrl(int c): Retorna verdadeiro se c for um caractere de controle ('\n', '\r', etc...);
    • int isxdigit(int c): Retorna verdadeiro se c for um dígito hexadecimal (0-9, A-F, a-f).

    Funções de conversão de caracteres¶

    • toupper(int c): Converte c para maiúscula, se for uma letra minúscula;
    • tolower(int c): Converte c para minúscula, se for uma letra maiúscula.

    stdlib.h¶

    A biblioteca (Standard Library Utilities) contém funções para alocação de memória, conversão de tipos, geração de números aleatórios e controle do fluxo do programa. Ela é uma das bibliotecas mais usadas na linguagem C.

    Funções de gerenciamento de memória dinâmica¶

    • void *malloc(size_t size);: Aloca size bytes de memória e retorna um ponteiro para o bloco alocado;
    • void *calloc(size_t num, size_t size);: Aloca e inicializa num elementos de size bytes com 0;
    • void *realloc(void *ptr, size_t new_size);: Redimensiona um bloco de memória alocado anteriormente;
    • void free(void *ptr);: Libera a memória alocada apontada por ptr.

    Funções de controle de fluxo do programa¶

    • void exit(int status);: Encerra o programa imediatamente com código de saída status;
    • void abort(void);: Termina o programa de forma anormal;
    • int atexit(void (*func)(void));: Registra uma função que será chamada ao encerrar o programa.

    Funções de ordenação e busca¶

    • void qsort(void *base, size_t num, size_t size, int (*compar)(const void *, const void *));: Ordena um arranjo genérico, armazenado na posição apontada pelo ponteiro base, com num elementos de size bytes cada um. A função de comparação usada é compar;
    • void *bsearch(const void *key, const void *base, size_t num, size_t size, int (*compar)(const void *, const void *));: Busca binária em um arranjo ordenado pelo valor key. O arranjo está armazenado na posição apontada pelo ponteiro base, com num elementos de size bytes cada um. A função de comparação usada é compar.

    #include <stdio.h>
#include <stdlib.h>

// Função de comparação para qsort() e bsearch()
int compara(const void *a, const void *b) {
    return (*(int*)a - *(int*)b);
}

int main() {
    int tamanho, chave;
    int *resultado;
    int numeros[] = {42, 23, 4, 16, 8, 15};

    tamanho = sizeof(numeros) / sizeof(numeros[0]);

    // Ordenando o array com qsort()
    qsort(numeros, tamanho, sizeof(int), compara);

    // Pedindo um número para buscar
    printf("Digite um número para buscar: ");
    scanf("%d", &chave);

    // Realizando a busca binária com bsearch()
    resultado = (int*) bsearch(&chave, numeros, tamanho, sizeof(int), compara);

    if (resultado != NULL) {
        printf("Número %d encontrado no array!\n", chave);
    } else {
        printf("Número %d não encontrado no array.\n", chave);
    }

    return 0;
}
